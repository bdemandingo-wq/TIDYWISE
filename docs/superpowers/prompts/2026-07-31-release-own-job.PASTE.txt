GOAL: let a cleaner release a job they claimed, so it returns to the Available
pool for someone else, with a time-based cutoff and an admin notification.

WHY AN RPC: both staff UPDATE policies on public.bookings have a WITH CHECK
requiring the new row's staff_id to be the caller's own staff record, so
staff_id -> NULL is rejected. Widening the policy would allow unassignment from
every update path in the app, and the cutoff must be enforced server-side.

CREATE OR REPLACE FUNCTION public.release_own_job(p_booking_id uuid)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_staff_id       uuid;
  v_booking        record;
  v_hours_until    numeric;
  v_notify_admin   boolean := false;
BEGIN
  -- 1. Caller must BE the assigned cleaner. Resolved from auth.uid(), never
  --    from a parameter, so this cannot be used to unassign someone else.
  SELECT b.* INTO v_booking
  FROM public.bookings b
  WHERE b.id = p_booking_id;

  IF v_booking.id IS NULL THEN
    RETURN jsonb_build_object('ok', false, 'reason', 'not_found');
  END IF;

  SELECT s.id INTO v_staff_id
  FROM public.staff s
  WHERE s.user_id = auth.uid()
    AND s.organization_id = v_booking.organization_id
    AND s.is_active = true
  ORDER BY s.created_at ASC
  LIMIT 1;

  IF v_staff_id IS NULL OR v_booking.staff_id IS DISTINCT FROM v_staff_id THEN
    RETURN jsonb_build_object('ok', false, 'reason', 'not_yours');
  END IF;

  -- 2. Refuse once the job has started. A released booking with a check-in on
  --    it and nobody assigned cannot be started by anyone else — the cleaner
  --    portal gates Start behind arrival — so it would sit stuck until an admin
  --    intervened. Better to refuse than to create that state.
  IF v_booking.status = 'in_progress' OR v_booking.cleaner_checkin_at IS NOT NULL THEN
    RETURN jsonb_build_object('ok', false, 'reason', 'already_started');
  END IF;

  IF v_booking.status <> 'confirmed' AND v_booking.status <> 'pending' THEN
    RETURN jsonb_build_object('ok', false, 'reason', 'wrong_status');
  END IF;

  -- 3. Time cutoff. scheduled_at is timestamptz so this is timezone-safe.
  --    Starting points, not gospel — change the two numbers here if they prove
  --    wrong in practice, and nothing else needs to move.
  v_hours_until := EXTRACT(EPOCH FROM (v_booking.scheduled_at - now())) / 3600.0;

  IF v_hours_until < 2 THEN
    RETURN jsonb_build_object('ok', false, 'reason', 'too_close',
                              'hours_until', round(v_hours_until, 1));
  ELSIF v_hours_until < 24 THEN
    v_notify_admin := true;   -- allowed, but the admin is told
  END IF;

  -- 4. Release. BOTH of these, in one transaction.
  UPDATE public.bookings
  SET staff_id = NULL
  WHERE id = p_booking_id;

  -- CRITICAL: syncCleanerPayShare writes a booking_team_assignments row even
  -- for single-cleaner bookings, and the payroll resolver reads pay_share
  -- FIRST, ahead of every other source. Leaving it behind would keep paying
  -- the cleaner who dropped the job.
  DELETE FROM public.booking_team_assignments
  WHERE booking_id = p_booking_id
    AND staff_id = v_staff_id;

  -- 5. Tell the admin. Always — not only inside the 24h window. A job going
  --    back in the pool is something they want to know about, and an
  --    unassigned job nobody notices at 7am is the failure this is avoiding.
  INSERT INTO public.admin_system_notifications
    (organization_id, type, title, message, link, metadata)
  VALUES (
    v_booking.organization_id,
    'staff_activity',
    CASE WHEN v_notify_admin
         THEN '⚠️ Job dropped close to start time'
         ELSE 'Job returned to Available' END,
    (SELECT COALESCE(s.name, 'A cleaner') FROM public.staff s WHERE s.id = v_staff_id)
      || ' released booking #' || COALESCE(v_booking.booking_number::text, left(p_booking_id::text, 8))
      || ' (' || to_char(v_booking.scheduled_at, 'Mon DD, HH24:MI') || ', '
      || round(v_hours_until) || 'h away). It is back in the Available pool.',
    '/dashboard/bookings',
    jsonb_build_object('booking_id', p_booking_id, 'released_by', v_staff_id,
                       'hours_until', round(v_hours_until, 1),
                       'short_notice', v_notify_admin)
  );

  RETURN jsonb_build_object('ok', true, 'short_notice', v_notify_admin,
                            'hours_until', round(v_hours_until, 1));
END;
$$;

REVOKE ALL ON FUNCTION public.release_own_job(uuid) FROM PUBLIC, anon;
GRANT EXECUTE ON FUNCTION public.release_own_job(uuid) TO authenticated;

COMMENT ON FUNCTION public.release_own_job(uuid) IS
'Lets a cleaner drop a job they claimed. SECURITY DEFINER because both staff
UPDATE policies on bookings have a WITH CHECK requiring staff_id to be the
caller''s own record, so staff_id -> NULL is rejected by RLS. Authorises
internally from auth.uid(); the booking id is the only parameter and cannot be
used to unassign anyone else.

Refuses once status = in_progress or cleaner_checkin_at is set: a released
booking carrying a check-in with nobody assigned cannot be started by another
cleaner, because the portal gates Start behind arrival.

DELETES the caller''s booking_team_assignments row as well as clearing
staff_id. syncCleanerPayShare writes that row even for single-cleaner bookings
and the payroll resolver reads pay_share first, so leaving it would keep paying
the cleaner who dropped the job.

Cutoff: free beyond 24h, allowed but admin-flagged between 2h and 24h, refused
inside 2h. Those two numbers are the only thing to change if the policy shifts.';
